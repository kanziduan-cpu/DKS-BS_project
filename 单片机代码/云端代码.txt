function normalizeEnvValue(rawValue) {
  const trimmed = String(rawValue || '').trim();

  if ((trimmed.startsWith('"') && trimmed.endsWith('"')) ||
      (trimmed.startsWith("'") && trimmed.endsWith("'"))) {
    return trimmed.slice(1, -1).trim();
  }

  return trimmed;
}

// ========== 原生读取 .env 文件（不依赖任何库，彻底解决 PM2 报错）==========
const fs = require('fs');
const path = require('path');

try {
  const envPath = path.resolve(__dirname, '.env');
  const envData = fs.readFileSync(envPath, 'utf8');
  
  envData.split('\n').forEach(line => {
    const trimmedLine = line.trim();
    if (!trimmedLine || trimmedLine.startsWith('#')) return;
    
    const [key, ...values] = trimmedLine.split('=');
    if (key) {
      process.env[key.trim()] = normalizeEnvValue(values.join('='));
    }
  });
} catch (err) {}

function normalizeCommandRecord(topic, payload, source, status = 'received') {
  const topicParts = topic.split('/');
  const topicDeviceId = topicParts[1] || DEFAULT_DEVICE_ID;
  const gatewayDeviceId = payload.gateway_device_id || payload.gatewayDeviceId || payload.device_id || payload.deviceId || topicDeviceId;
  const targetDeviceId = payload.target_device_id || payload.targetDeviceId || topicDeviceId || gatewayDeviceId;

  return {
    topic,
    source,
    status,
    gateway_device_id: gatewayDeviceId,
    target_device_id: targetDeviceId,
    payload,
    queued_at: new Date().toISOString()
  };
}

function rememberCommandAudit(record) {
  commandAuditCache.unshift(record);
  if (commandAuditCache.length > HISTORY_LIMIT) {
    commandAuditCache.length = HISTORY_LIMIT;
  }
}

function enqueuePendingCommand(record) {
  const queue = pendingCommandQueues.get(record.gateway_device_id) || [];
  queue.push(record);
  pendingCommandQueues.set(record.gateway_device_id, queue);
}

function dequeuePendingCommand(deviceId) {
  const queue = pendingCommandQueues.get(deviceId) || [];
  const record = queue.shift() || null;

  if (queue.length > 0) {
    pendingCommandQueues.set(deviceId, queue);
  } else {
    pendingCommandQueues.delete(deviceId);
  }

  return record;
}

  const latestSensor = latestSensorCache.get(deviceId);
  if (latestSensor && latestSensor.cloud_connected !== undefined && latestSensor.cloud_connected !== null) {

  const latestStatus = latestStatusCache.get(deviceId);
  if (latestStatus && latestStatus.cloud_connected !== undefined && latestStatus.cloud_connected !== null) {

  return null;
}

function safeJsonParse(buffer) {
  try {
    return JSON.parse(buffer.toString());
  } catch (error) {
    console.error('JSON parse failed:', error.message);

}

function toNumber(value, fallback = null) {
  if (value === null || value === undefined || value === '') {
    return fallback;
  }
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
}

function toIsoTimestamp(value, fallback = new Date()) {
  if (value === null || value === undefined || value === '') {
    return fallback.toISOString();
  }

  if (value instanceof Date && !Number.isNaN(value.getTime())) {
    return value.toISOString();
  }

  const numericValue = Number(value);
  if (Number.isFinite(numericValue)) {
    if (numericValue > 1e12) {
      return new Date(numericValue).toISOString();
    }

    if (numericValue > 1e9) {
      return new Date(numericValue * 1000).toISOString();
    }
  }

  const parsed = new Date(value);
  if (!Number.isNaN(parsed.getTime())) {
    return parsed.toISOString();
  }

  return fallback.toISOString();
}

function isMqttConnected() {
  return Boolean(mqttClient && mqttClient.connected);
}

function formatMqttError(error) {
  if (!error) {
    return 'unknown mqtt error';
  }

  const parts = [
    error.message,
    error.code,
    error.errno,
    error.syscall,
    error.address,
    error.port
  ].filter((value) => value !== undefined && value !== null && value !== '');

  return parts.length ? parts.join(' | ') : JSON.stringify(error);
}

function publishHttpBridgeToMqtt(topic, payload, event) {
  if (!isMqttConnected()) {
    return;
  }

  mqttClient.publish(topic, JSON.stringify({
    ...payload,
    bridge_origin: 'http-api'
  }), { qos: 1 }, (error) => {
    if (error) {
      console.error('http bridge publish failed:', error.message);
      return;
    }

    logEvent(event, {
      topic,
      device_id: payload.device_id || payload.deviceId || null
    });
  });
}

function vibrationLevelToAppValue(level) {
  switch (Number(level)) {
    case 3:
      return 90;
    case 2:
      return 70;
    case 1:
      return 30;
    default:
      return 0;
  }
}

function normalizeDeviceStatusPayload(topic, payload) {
  const topicParts = topic.split('/');
  const topicDeviceId = topicParts[1] || DEFAULT_DEVICE_ID;
  const params = payload.params || payload;

  return {
    device_id: params.device_id || payload.device_id || topicDeviceId,
    machine_code: params.machine_code || payload.machine_code || null,
    is_online: toBoolean(params.is_online ?? payload.is_online),
    is_running: toBoolean(params.is_running ?? payload.is_running),
    wifi_connected: toBoolean(params.wifi_connected ?? payload.wifi_connected),
    cloud_connected: toBoolean(params.cloud_connected ?? payload.cloud_connected),
    firmware_version: params.firmware_version || payload.firmware_version || null,

    reported_at: (() => {
      const val = params.reported_at ?? payload.reported_at;
      if (!val) return new Date().toISOString();

      return Number.isFinite(d.getTime()) ? d.toISOString() : new Date().toISOString();
    })(),
    updated_at: new Date().toISOString()
  };
}

    id: String(alarm.id || `alarm_${alarm.device_id}_${Date.now()}`),
    warehouseId: alarm.warehouse_id || DEFAULT_WAREHOUSE_ID,
    deviceId: alarm.device_id,
    type: alarm.alarm_type,

    alarmTitle: alarm.alarm_title || '仓库环境告警',

    thresholdValue: Number(alarm.threshold_value || 0),
    status: alarm.status || 'UNPROCESSED',
    timestamp: new Date(createdAt).getTime()
  };
}

      warehouse_id: DEFAULT_WAREHOUSE_ID,
      device_id: cleanData.device_id,
      alarm_type: 'TEMPERATURE_HIGH',
      level: 'warning',
      alarm_title: '温度过高',
      alarm_message: `设备 ${cleanData.device_id} 温度过高`,
      alarm_value: String(cleanData.temperature),
      threshold_value: thresholds.temperature.max,
      status: 'UNPROCESSED',
      created_at: new Date().toISOString()
    });
  }

  if (cleanData.temperature != null && cleanData.temperature < thresholds.temperature.min) {
    alarms.push({
      warehouse_id: DEFAULT_WAREHOUSE_ID,
      device_id: cleanData.device_id,

async function insertAlarms(alarms) {
  if (!alarms.length) {
    return [];
  }

  const inserted = [];
  for (const alarm of alarms) {
    const { data: existing } = await supabase
      .from('alarms')
      .select('id')
      .eq('device_id', alarm.device_id)
      .eq('alarm_type', alarm.alarm_type)
      .eq('status', 'UNPROCESSED')
      .limit(1);

    if (existing && existing.length > 0) {
      continue;
    }

    const { data, error } = await supabase.from('alarms').insert(alarm).select();
    if (error) {
      console.error('insertAlarms failed:', error.message);
    } else if (data) {
      inserted.push(...data);
    }
  }
  return inserted;
}

function publishAppAlarm(alarm) {
  const payload = toAppAlarmPayload(alarm);
  rememberAlarm(payload);

  if (!isMqttConnected()) {
    return;
  }

  const topic = `warehouse/${alarm.device_id}/alarm`;
  mqttClient.publish(topic, JSON.stringify(payload), { qos: 1 });
}

function buildMcuCommand(body) {
  const gatewayDeviceId = body.gatewayDeviceId || body.gateway_device_id || DEFAULT_DEVICE_ID;
  const targetDeviceId = body.targetDeviceId || body.target_device_id || body.deviceId || body.device_id || gatewayDeviceId;
  const action = String(body.action || body.command || '').trim().toLowerCase();

  if (!action) {
    throw new Error('missing action');
  }

  if (action === 'control_servo' || action === 'set_angle' || action === 'set_servo_angle') {
    return {
      topic: `warehouse/${gatewayDeviceId}/command`,
      payload: {
        device_id: gatewayDeviceId,
        target_device_id: targetDeviceId,
        servo_angle: Number(body.value ?? body.angle ?? body?.params?.angle ?? 0),
        timestamp: Date.now()
      }
    };
  }

  if (action === 'enable_auto' || action === 'auto_on') {
    return {
      topic: `warehouse/${gatewayDeviceId}/command`,
      payload: {
        device_id: gatewayDeviceId,
        target_device_id: targetDeviceId,
        auto_control_enabled: true,
        timestamp: Date.now()
      }
    };
  }

  if (action === 'disable_auto' || action === 'auto_off') {
    return {
      topic: `warehouse/${gatewayDeviceId}/command`,
      payload: {
        device_id: gatewayDeviceId,
        target_device_id: targetDeviceId,
        auto_control_enabled: false,
        timestamp: Date.now()
      }
    };
  }

  if (action === 'enable_buzzer' || action === 'buzzer_on') {
    return {
      topic: `warehouse/${gatewayDeviceId}/command`,
      payload: {
        device_id: gatewayDeviceId,
        target_device_id: targetDeviceId,
        buzzer_enabled: true,
        timestamp: Date.now()
      }
    };
  }

  if (action === 'disable_buzzer' || action === 'buzzer_off') {
    return {
      topic: `warehouse/${gatewayDeviceId}/command`,
      payload: {
        device_id: gatewayDeviceId,
        target_device_id: targetDeviceId,
        buzzer_enabled: false,
        timestamp: Date.now()
      }
    };
  }

  return {
    topic: `warehouse/${gatewayDeviceId}/command`,
    payload: {
      device_id: gatewayDeviceId,
      target_device_id: targetDeviceId,
      action: action.toUpperCase(),
      value: String(body.value ?? 1),
      timestamp: Date.now()
    }
  };
}

async function getLatestReading(deviceId) {
  const { data, error } = await supabase
    .from('readings')
    .select('*')
    .eq('device_id', deviceId)
    .order('created_at', { ascending: false })
    .limit(1);

  if (error) {
    console.error('getLatestReading failed:', error.message);
    return latestSensorCache.get(deviceId) || null;
  }

  return data?.[0] || latestSensorCache.get(deviceId) || null;
}

async function getAlarmList(deviceId, status, limit) {
  let query = supabase
    .from('alarms')
    .select('*')
    .order('created_at', { ascending: false })
    .limit(limit);

  if (deviceId) {
    query = query.eq('device_id', deviceId);
  }
  if (status) {
    query = query.eq('status', status);
  }

  const { data, error } = await query;
  if (error) {
    console.error('getAlarmList failed:', error.message);
    return alarmCache
      .filter((alarm) => (!deviceId || alarm.deviceId === deviceId) && (!status || alarm.status === status))
      .slice(0, limit);
  }

  return (data || []).map(toAppAlarmPayload);
}

if (mqttClient) {
  mqttClient.on('connect', () => {
    console.log(`✅ MQTT 已连接到 Broker: mqtt://${MQTT_HOST}:${MQTT_PORT}`);
    mqttClient.subscribe([
      'warehouse/+/sensor/data',
      'warehouse/+/device/status',
      'warehouse/+/command'
    ], (error) => {
      if (error) {
        console.error('订阅失败:', error.message);
        return;
      }
      console.log('👂 正在监听 [单片机] 传感器数据、设备状态与控制指令...');
    });
  });

  mqttClient.on('error', (error) => {
    console.error('MQTT 错误:', formatMqttError(error));
  });

  mqttClient.on('offline', () => {
    console.warn(`MQTT 客户端离线: mqtt://${MQTT_HOST}:${MQTT_PORT}`);
  });

  mqttClient.on('message', async (topic, message) => {
    const payload = safeJsonParse(message);
    if (!payload) {
      return;
    }

    if (payload.bridge_origin === 'http-api') {
      logEvent('mqtt_bridge_message_ignored', { topic });
      return;
    }

    logEvent('mqtt_message_received', { topic, payload });

    if (topic.endsWith('/sensor/data')) {
      const cleanData = normalizeSensorPayload(topic, payload);
      rememberSensorData(cleanData);
      await insertReading(cleanData);
      logEvent('sensor_data_processed', summarizeSensorData(cleanData));

      const alarms = await insertAlarms(buildAlarmRecords(cleanData));
      alarms.forEach(publishAppAlarm);
      return;
    }

    if (topic.endsWith('/device/status')) {
      const statusData = normalizeDeviceStatusPayload(topic, payload);
      rememberDeviceStatus(statusData);
      await upsertDeviceStatus(statusData);
      logEvent('device_status_processed', summarizeStatusData(statusData));
      return;
    }

    if (topic.endsWith('/command')) {
      const record = normalizeCommandRecord(topic, payload, 'mqtt', 'broker-observed');
      const topicDeviceId = topic.split('/')[1] || DEFAULT_DEVICE_ID;
      const needsGatewayRelay = topicDeviceId !== record.gateway_device_id;
      const deviceCloudConnected = getDeviceCloudConnected(record.gateway_device_id);
      const queueForHttpFallback = deviceCloudConnected === false && !needsGatewayRelay;

      rememberCommandAudit(record);

      if (queueForHttpFallback) {
        enqueuePendingCommand({ ...record, status: 'queued' });
        logEvent('command_queued_from_mqtt_for_http_fallback', {
          gateway_device_id: record.gateway_device_id,
          target_device_id: record.target_device_id,
          topic: record.topic,
          deviceCloudConnected
        });
      }

      if (needsGatewayRelay && isMqttConnected()) {
        const relayTopic = `warehouse/${record.gateway_device_id}/command`;
        mqttClient.publish(relayTopic, JSON.stringify(record.payload), { qos: 1 }, (error) => {
          if (error) {
            console.error('command relay failed:', error.message);
            return;
          }

          logEvent('command_relayed_to_gateway_topic', {
            sourceTopic: topic,
            relayTopic,
            gateway_device_id: record.gateway_device_id,
            target_device_id: record.target_device_id
          });
        });
      }

      logEvent('command_observed', {
        topic,
        payload,
        gateway_device_id: record.gateway_device_id,
        target_device_id: record.target_device_id,
        queueForHttpFallback,
        needsGatewayRelay
      });
    }
  });
} else {
  console.warn('MQTT 已禁用，服务端将以 HTTP-only 模式运行。');
}

app.post('/api/control/command', (req, res) => {
  try {
    const command = buildMcuCommand(req.body || {});
    const deviceCloudConnected = getDeviceCloudConnected(command.payload.device_id);
    const queueForHttpFallback = !MQTT_ENABLED || !isMqttConnected() || deviceCloudConnected === false;
    const record = normalizeCommandRecord(command.topic, command.payload, 'http-api', queueForHttpFallback ? 'queued' : 'published');

    rememberCommandAudit(record);

    if (queueForHttpFallback) {
      enqueuePendingCommand(record);
      logEvent('command_queued_for_http_fallback', {
        device_id: record.gateway_device_id,
        target_device_id: record.target_device_id,
        topic: record.topic,
        mqttEnabled: MQTT_ENABLED,
        mqttConnected: isMqttConnected(),
        deviceCloudConnected
      });
    }

    if (MQTT_ENABLED && isMqttConnected()) {
      mqttClient.publish(command.topic, JSON.stringify(command.payload), { qos: 1 }, (error) => {
        if (error) {
          console.error('指令发布失败:', error.message);
          return res.status(500).json({ success: false, message: error.message });
        }

        logEvent('command_published', {
          topic: command.topic,
          payload: command.payload,
          queuedForFallback: queueForHttpFallback
        });

        res.json({
          success: true,
          topic: command.topic,
          payload: command.payload,
          queued: queueForHttpFallback,
          mqttDelivered: true
        });
      });
      return;
    }

    res.json({
      success: true,
      topic: command.topic,
      payload: command.payload,
      queued: true,
      mqttDelivered: false,
      message: '命令已入队，等待设备通过 HTTP 降级链路轮询获取'
    });
  } catch (error) {
    res.status(400).json({ success: false, message: error.message });
  }
});
